        "    return int(m.group(1)) if m else None\n",

        "    \"\"\"Greedy splitter that prefers paragraph/sentence boundaries, falling\n",
        "    back to hard character cuts — a dependency-free stand-in for LangChain's\n",
        "    RecursiveCharacterTextSplitter.\"\"\"\n",
        "    separators = [\"\\n\\n\", \"\\n\", \". \", \" \"]\n",
        "\n",
        "    def _split(t, seps):\n",
        "        if len(t) <= chunk_size:\n",
        "            return [t]\n",
        "        if not seps:\n",
        "            return [t[i:i + chunk_size] for i in range(0, len(t), chunk_size)]\n",
        "        sep, rest = seps[0], seps[1:]\n",
        "        parts = t.split(sep)\n",
        "        chunks, current = [], \"\"\n",
